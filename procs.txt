/*
    add(Open)   # 1
    add(write)  # 2
    add(read)   # ...
    add(lambda t: t.pop().flush())
*/

print:
begin
  @str stdout @write
  "\n" stdout @write
  stdout @flush
end

write:
begin
  @write
end

swap: a b
begin
  b a
end

dup: x
begin
  x x
end

dprint:begin
  dup print
end

nprint:begin
  repeat print end
end

dropStack: begin
  // top = how many elements are on the stack
  do // while top of the stack isn't 0
    pop
    top// push new element count
  while
end

rot: a b c begin
  b c a
end

rrot: a b c begin
  c a b
end

over: a b
begin
  a b a
end
